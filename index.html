function renderThumbnails() { imageList.innerHTML = ""; images.forEach((item, index) => { const div = document.createElement("div"); div.className = "thumbnail"; div.draggable = true; div.dataset.index = index; div.innerHTML = ` `; imageList.appendChild(div); }); /* * ========================= * REMOVE IMAGE * ========================= */ document .querySelectorAll(".remove-image") .forEach(button => { button.addEventListener( "click", function(event) { event.stopPropagation(); const index = Number( this.dataset.index ); images.splice( index, 1 ); renderThumbnails(); renderCanvas(); } ); }); /* * ========================= * DRAG & DROP * ========================= */ document .querySelectorAll(".thumbnail") .forEach(thumbnail => { /* * START DRAG */ thumbnail.addEventListener( "dragstart", function(event) { this.classList.add( "dragging" ); event.dataTransfer.effectAllowed = "move"; event.dataTransfer.setData( "text/plain", this.dataset.index ); } ); /* * END DRAG */ thumbnail.addEventListener( "dragend", function() { this.classList.remove( "dragging" ); document .querySelectorAll( ".thumbnail" ) .forEach(item => { item.classList.remove( "drag-over" ); }); } ); /* * DRAG OVER */ thumbnail.addEventListener( "dragover", function(event) { event.preventDefault(); event.dataTransfer.dropEffect = "move"; this.classList.add( "drag-over" ); } ); /* * DRAG LEAVE */ thumbnail.addEventListener( "dragleave", function() { this.classList.remove( "drag-over" ); } ); /* * DROP */ thumbnail.addEventListener( "drop", function(event) { event.preventDefault(); this.classList.remove( "drag-over" ); const fromIndex = Number( event.dataTransfer .getData( "text/plain" ) ); const toIndex = Number( this.dataset.index ); if ( Number.isNaN( fromIndex ) ) { return; } if ( fromIndex === toIndex ) { return; } /* * Move image */ const movedImage = images.splice( fromIndex, 1 )[0]; images.splice( toIndex, 0, movedImage ); /* * Rebuild thumbnails * and update collage */ renderThumbnails(); renderCanvas(); } ); }); }
